import { View, Text } from 'react-native'
import { useForm } from '@tanstack/react-form'
import { useQueryClient } from '@tanstack/react-query'
import { useTranslation } from 'react-i18next'
import { Haptics } from 'react-native-nitro-haptics'
import { SignInBody } from '@template/client-typescript/typescript'
import { KeyboardAware } from '@/components/ui/KeyboardAware'
import { Button } from '@/components/ui/Button'
import { Input, InputGroup } from '@/components/ui/Input'
import type { DeepPartial } from '@/lib/types'
import * as AppleAuthentication from 'expo-apple-authentication'

type SignInFormInput = typeof SignInBody['_output']

interface SignInFormProps {
	onSuccess?: () => void
}

export function SignInForm({ onSuccess }: SignInFormProps) {
	const { t } = useTranslation()
	const queryClient = useQueryClient()

	const form = useForm({
		defaultValues: {} as DeepPartial<SignInFormInput>,
		validators: { onChange: SignInBody },
		onSubmit: async ({ value }) => {
			const result = SignInBody.safeParse(value)
			if (!result.success) return
			// TODO: call the SDK mutation hook with result.data and surface errors via toast.
			Haptics.notification('success')
			// TODO: replace with the real list query key: queryClient.invalidateQueries({ queryKey: listSignInFormQueryKey() })
			await queryClient.invalidateQueries({ queryKey: [] })
			onSuccess?.()
		},
	})

	const handleAppleSignIn = async () => {
		// 1) Native Apple Sign-In returns an identity token signed by Apple.
		const credential = await AppleAuthentication.signInAsync({
			requestedScopes: [
				AppleAuthentication.AppleAuthenticationScope.FULL_NAME,
				AppleAuthentication.AppleAuthenticationScope.EMAIL,
			],
		})
		if (!credential.identityToken) throw new Error('Apple did not return an identity token')
		// 2) Hand the token to your auth helper (see packages/app/expo/lib/auth).
		// TODO: integrate with the form payload as appropriate.
	}

	return (
		<KeyboardAware className="flex-1">
			<View className="px-5 pt-6 gap-4">
				{/* TODO: render <form.Field> entries here. Example:
				<form.Field name="example">
					{field => (
						<View>
							<InputGroup>
								<Input
									value={field.state.value ?? ''}
									onChangeText={field.handleChange}
									onBlur={field.handleBlur}
									placeholder={t('auth.signIn.fields.example.placeholder')}
								/>
							</InputGroup>
							{!field.state.meta.isValid && (
								<Text className="text-accent-danger font-sans">{String(field.state.meta.errors[0] ?? '')}</Text>
							)}
						</View>
					)}
				</form.Field> */}
			</View>

			<form.Subscribe selector={s => ({ canSubmit: s.canSubmit, isSubmitting: s.isSubmitting, values: s.values })}>
				{({ canSubmit, isSubmitting, values }) => {
					const isDisabled = !canSubmit || isSubmitting || !SignInBody.safeParse(values).success
					return (
						<View className="px-5 pt-2 pb-6 gap-3">
							<AppleAuthentication.AppleAuthenticationButton
								buttonType={AppleAuthentication.AppleAuthenticationButtonType.SIGN_IN}
								buttonStyle={AppleAuthentication.AppleAuthenticationButtonStyle.WHITE}
								cornerRadius={9999}
								style={{ width: '100%', height: 52 }}
								onPress={() => void handleAppleSignIn()}
							/>
							<Button
								fullWidth
								label={t('auth.signIn.submit')}
								onPress={() => void form.handleSubmit()}
								disabled={isDisabled}
							/>
						</View>
					)
				}}
			</form.Subscribe>
		</KeyboardAware>
	)
}
