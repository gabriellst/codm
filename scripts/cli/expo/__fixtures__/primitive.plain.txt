import { View, type ViewProps } from 'react-native'
import { cva, type VariantProps } from 'class-variance-authority'
import { cn } from '@/lib/utils'

const spacerVariants = cva('', {
	variants: {
		variant: {
			default: '',
			secondary: '',
		},
	},
	defaultVariants: { variant: 'default' },
})

interface SpacerProps extends ViewProps, VariantProps<typeof spacerVariants> {}

export function Spacer({ className, variant, ...props }: SpacerProps) {
	return <View className={cn(spacerVariants({ variant }), className)} {...props} />
}

export { spacerVariants }
