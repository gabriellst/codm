import { Text, View } from 'react-native'
import { router } from 'expo-router'
import { useForm } from '@tanstack/react-form'
import { useTranslation } from 'react-i18next'
import { KeyboardAware } from '@/components/ui/KeyboardAware'
import { Button } from '@/components/ui/Button'
import { updateBioRequestSchema } from '@codedm/client-typescript/typescript'
import type { DeepPartial } from '@/lib/types'

type EditBioInput = typeof updateBioRequestSchema['_output']

export default function EditBioSheet() {
	const { t } = useTranslation()

	const form = useForm({
		defaultValues: {} as DeepPartial<EditBioInput>,
		validators: { onChange: updateBioRequestSchema },
		onSubmit: async ({ value }) => {
			const result = updateBioRequestSchema.safeParse(value)
			if (!result.success) return
			// TODO: call SDK mutation and on success dismiss the sheet.
			router.back()
		},
	})

	return (
		<KeyboardAware className="flex-1">
			<View className="flex-1 px-5 pt-4 pb-6 gap-4">
				<Text className="text-fg-0 font-sans-bold text-xl">{t('profile.editBio.title')}</Text>
				{/* TODO: render <form.Field> entries here. */}
			</View>

			<form.Subscribe selector={s => ({ canSubmit: s.canSubmit, isSubmitting: s.isSubmitting })}>
				{({ canSubmit, isSubmitting }) => (
					<View className="px-5 pb-6">
						<Button
							fullWidth
							label={t('profile.editBio.submit')}
							onPress={() => void form.handleSubmit()}
							disabled={!canSubmit || isSubmitting}
						/>
					</View>
				)}
			</form.Subscribe>
		</KeyboardAware>
	)
}
