import { View, Text } from 'react-native'
import { useForm } from '@tanstack/react-form'
import { useQueryClient } from '@tanstack/react-query'
import { useTranslation } from 'react-i18next'
import { Haptics } from 'react-native-nitro-haptics'
import { UpdateProfileBody } from '@codedm/client-typescript/typescript'
import { KeyboardAware } from '@/components/ui/KeyboardAware'
import { Button } from '@/components/ui/Button'
import { Input, InputGroup } from '@/components/ui/Input'
import type { DeepPartial } from '@/lib/types'

type ProfileEditFormInput = typeof UpdateProfileBody['_output']

interface ProfileEditFormProps {
	onSuccess?: () => void
}

export function ProfileEditForm({ onSuccess }: ProfileEditFormProps) {
	const { t } = useTranslation()
	const queryClient = useQueryClient()

	const form = useForm({
		defaultValues: {} as DeepPartial<ProfileEditFormInput>,
		validators: { onChange: UpdateProfileBody },
		onSubmit: async ({ value }) => {
			const result = UpdateProfileBody.safeParse(value)
			if (!result.success) return
			// TODO: call the SDK mutation hook with result.data and surface errors via toast.
			Haptics.notification('success')
			// TODO: replace with the real list query key: queryClient.invalidateQueries({ queryKey: listProfileEditFormQueryKey() })
			await queryClient.invalidateQueries({ queryKey: [] })
			onSuccess?.()
		},
	})

	return (
		<KeyboardAware className="flex-1">
			<View className="px-5 pt-6 gap-4">
				{/* TODO: render <form.Field> entries here. Example:
				<form.Field name="example">
					{field => (
						<View>
							<InputGroup>
								<Input
									value={field.state.value ?? ''}
									onChangeText={field.handleChange}
									onBlur={field.handleBlur}
									placeholder={t('profile.edit.fields.example.placeholder')}
								/>
							</InputGroup>
							{!field.state.meta.isValid && (
								<Text className="text-accent-danger font-sans">{String(field.state.meta.errors[0] ?? '')}</Text>
							)}
						</View>
					)}
				</form.Field> */}
			</View>

			<form.Subscribe selector={s => ({ canSubmit: s.canSubmit, isSubmitting: s.isSubmitting, values: s.values })}>
				{({ canSubmit, isSubmitting, values }) => {
					const isDisabled = !canSubmit || isSubmitting || !UpdateProfileBody.safeParse(values).success
					return (
						<View className="px-5 pt-2 pb-6 gap-3">
							<Button
								fullWidth
								label={t('profile.edit.submit')}
								onPress={() => void form.handleSubmit()}
								disabled={isDisabled}
							/>
						</View>
					)
				}}
			</form.Subscribe>
		</KeyboardAware>
	)
}
