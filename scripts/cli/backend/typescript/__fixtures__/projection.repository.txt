/**
 * Read + projection-write surface for MessageProjection.
 *
 * Canonical (mandatory) methods:
 *   - findByKey: read half of `find → applyEvent → save`
 *   - save:      write half (upsert by natural key)
 *   - insertIfNew: replay-safe creation (ON CONFLICT DO NOTHING)
 *
 * Atomic ops are EDGE CASES — add one only when a canonical-flow trigger fails:
 *   • hot-row contention (concurrent writes to the same row)
 *   • bulk operation (one event mutates N rows)
 *   • monotonic constraint (only-if-greater)
 *   • conditional update (requires SQL atomicity)
 *   • cache-mirror bulk upsert (archetype A / B1)
 *
 * Each atomic method should carry a comment naming the trigger that justifies it.
 */
export abstract class MessageProjectionRepository {
	// ── Canonical (always required) ──────────────────────────────────────
	abstract findByKey(/* key: TODOKey */ key: unknown, tx?: Transaction): Promise<MessageProjection | null>
	abstract save(projection: MessageProjection, tx?: Transaction): Promise<void>
	abstract insertIfNew(projection: MessageProjection, tx?: Transaction): Promise<boolean>

	// ── Atomic ops (edge cases — justify each one in a comment) ──────────
	// abstract markXyzMany(ids: string[], at: Date, tx?: Transaction): Promise<void>
}
