import { BaseEntity, z } from '@codm/core-typescript'
import Z from 'zod'
import { DomainErrors } from '../errors'

export const OrderSchema = z.object({
	// Define entity properties
	name: z.string().min(1, { error: 'TODO_NAME_REQUIRED' as DomainErrors }),
	// Use ids for entity relation:
	// ownerId: z.instance(Id),
})

export type OrderProps = Z.infer<typeof OrderSchema>

export class Order extends BaseEntity<typeof OrderSchema> {
	static override schema = OrderSchema

	static create(data: { name: string }): Order {
		return new Order({
			name: data.name.trim(),
		})
	}

	// Mutation methods:

	// }
}
