// packages/app/expo/app/active-exercise/-types.ts
//
// The form-data schema is the API mutation request PLUS one UI-only field
// (`editingSelection`) that records which row, if any, the user is editing.
// Submit-time we re-parse with the bare API schema to strip it.
//
// `SetEditorForm` is exported as a TYPE only — `_inferSetEditorForm` is a
// function but it's never invoked at runtime. Its presence lets every
// downstream component type its `form` prop without restating TanStack
// Form's generic arguments at every call site.

import { z } from 'zod/v4'
import { useForm } from '@tanstack/react-form'
import { logSetMutationRequestSchema } from '@template/client-typescript/typescript'
import type { DeepPartial } from '@/lib'

/**
 * `null` = log mode (the user is shaping the next set).
 * Populated = edit mode for that specific persisted set.
 */
const editingSelectionSchema = z
	.object({
		workoutSetId: z.string(),
		revisionNumber: z.number().int().nonnegative(),
	})
	.nullable()

export type EditingSelection = z.infer<typeof editingSelectionSchema>

/**
 * Form schema = API mutation request + UI-only `editingSelection`.
 * Validators run against this merged shape; submit-time we re-parse the
 * API subset with `logSetMutationRequestSchema` to strip the UI field.
 */
export const setEditorFormSchema = logSetMutationRequestSchema.and(
	z.object({ editingSelection: editingSelectionSchema }),
)

export type SetEditorFormData = z.infer<typeof setEditorFormSchema>

/**
 * Type-only inference helper. NEVER invoked at runtime — exists solely
 * so consumer components can type their `form` prop as `SetEditorForm`
 * without re-deriving the form's generic args.
 */
function _inferSetEditorForm() {
	const defaults: DeepPartial<SetEditorFormData> = {}
	return useForm({
		defaultValues: defaults,
		validators: { onChange: setEditorFormSchema },
	})
}
export type SetEditorForm = ReturnType<typeof _inferSetEditorForm>
