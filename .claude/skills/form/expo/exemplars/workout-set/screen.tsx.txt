// packages/app/expo/app/active-exercise/[id].tsx
//
// The screen fetches the exercise and GATES the form's mount on it. The
// form's reps/kg defaults seed from `data.suggestedNextReps/Kg`, and
// `defaultValues` is read only ONCE at mount — it does NOT auto-sync when
// the query resolves later. So a form created before `data` arrives would
// seed 0/0 and never pick up the suggestion. The inner `ActiveExerciseForm`
// — which owns the single `useForm` threaded to the editor + list — mounts
// only once `data` exists. (Static chrome can render above the gate.)

import { View } from 'react-native'
import { useLocalSearchParams } from 'expo-router'
import { useForm } from '@tanstack/react-form'
import { useQueryClient } from '@tanstack/react-query'
import {
	useGetActiveExercise,
	useLogSet,
	useEditSet,
	logSetMutationRequestSchema,
	getActiveExerciseQueryKey,
	type GetActiveExercise200,
	type LogSetMutationRequest,
} from '@codedm/client-typescript/typescript'
import type { DeepPartial } from '@/lib'
import { SetEditor } from './-components/SetEditor'
import { SetList } from './-components/SetList'
import { SetsSkeleton } from './-components/SetsSkeleton'
import { setEditorFormSchema, type EditingSelection, type SetEditorFormData } from './-types'

export default function ActiveExerciseScreen() {
	const { id } = useLocalSearchParams<{ id: string }>()
	const { data } = useGetActiveExercise({ workoutExerciseId: id })

	// Gate-then-mount: the form's defaults seed from `data`, so it must not
	// mount until `data` exists.
	if (!data) return <SetsSkeleton />

	return <ActiveExerciseForm exerciseId={id} data={data} />
}

function ActiveExerciseForm({ exerciseId, data }: { exerciseId: string; data: GetActiveExercise200 }) {
	const queryClient = useQueryClient()
	const logSet = useLogSet()
	const editSet = useEditSet()

	const performLog = (req: LogSetMutationRequest) =>
		logSet.mutateAsync({ workoutExerciseId: exerciseId, data: req })

	const performEdit = (sel: NonNullable<EditingSelection>, req: LogSetMutationRequest) =>
		editSet.mutateAsync({
			workoutExerciseId: exerciseId,
			workoutSetId: sel.workoutSetId,
			data: { expectedRevisionNumber: sel.revisionNumber, ...req },
		})

	// One form instance for the whole screen, seeded from real data. Children
	// receive it as a typed prop — no Zustand mirror, no Context.
	const form = useForm({
		defaultValues: {
			reps: data.suggestedNextReps,
			kg: data.suggestedNextKg,
			isDropset: false,
			editingSelection: null,
		} as DeepPartial<SetEditorFormData>,
		validators: { onChange: setEditorFormSchema },
		onSubmit: async ({ value }) => {
			// Strip the UI-only field by re-parsing with the bare API schema.
			const api = logSetMutationRequestSchema.safeParse({
				reps: value.reps,
				kg: value.kg,
				isDropset: value.isDropset,
			})
			if (!api.success) return

			// Branch on the UI-only field — populated → edit, null → log.
			const sel = value.editingSelection
			await (sel != null ? performEdit(sel, api.data) : performLog(api.data))

			await queryClient.invalidateQueries({
				queryKey: getActiveExerciseQueryKey({ workoutExerciseId: exerciseId }),
			})
			form.reset()
		},
	})

	return (
		<View className="flex-1">
			{/* Both children share the same `form` instance — every change in
			    SetEditor's NumFields re-renders SetRow's 3-mode selector in
			    the same React commit. No microtask hop. */}
			<SetList form={form} sets={data.sets} workoutExerciseId={exerciseId} />
			<SetEditor form={form} />
		</View>
	)
}
