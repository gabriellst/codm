// packages/app/expo/app/active-exercise/-components/SetList/SetRow/index.tsx
//
// The 3-mode discriminant selector is the keystone of this whole pattern.
//
//   selector returns 0 | 1 | 2 as const   (not an object!)
//
//     0 → no row is currently selected for editing
//     1 → THIS row is the one being edited
//     2 → ANOTHER row is being edited (dim this row)
//
// Because the selector returns a NUMBER, TanStack Form's structural-equality
// check only triggers a re-render when the mode actually changes — typing in
// the reps NumField does NOT re-render every row. If we returned an object
// `{ isSelected, isOther }`, every keystroke would create a new object and
// invalidate the equality check.

import { Pressable, Text, View, type PressableProps } from 'react-native'
import { useTranslation } from 'react-i18next'
import type { GetActiveExercise200 } from '@codedm/client-typescript/typescript'
import type { SetEditorForm } from '@/app/active-exercise/-types'

type ActiveSet = GetActiveExercise200['sets'][number]

interface SetRowProps extends Omit<PressableProps, 'children' | 'onPress'> {
	form: SetEditorForm
	set: ActiveSet
}

export function SetRow({ form, set, ...rest }: SetRowProps) {
	const { t } = useTranslation()

	// Four writes inside one handler. React 18 batches them into ONE
	// commit, so SetEditor's subscriber sees the full edit-mode payload
	// in a single render. Each setFieldValue marks the field as touched —
	// a concurrent server refetch cannot stomp these values.
	const handleSelect = () => {
		form.setFieldValue('editingSelection', {
			workoutSetId: set.workoutSetId,
			revisionNumber: set.revisionNumber,
		})
		form.setFieldValue('reps', set.reps)
		form.setFieldValue('kg', set.kg)
		form.setFieldValue('isDropset', set.isDropset)
	}

	return (
		<form.Subscribe
			selector={s => {
				const sel = s.values.editingSelection
				if (sel == null) return 0 as const
				return sel.workoutSetId === set.workoutSetId ? (1 as const) : (2 as const)
			}}
		>
			{mode => {
				const isSelected = mode === 1
				const isOtherEditing = mode === 2

				return (
					<Pressable
						onPress={handleSelect}
						accessibilityRole="button"
						accessibilityLabel={t('workout.setActions.edit')}
						className={[
							'flex-row items-center gap-2.5 py-4 px-3.5 border rounded-md',
							isOtherEditing ? 'opacity-40' : '',
							isSelected
								? 'bg-surface-3/40 border-white/[0.32]'
								: 'bg-surface-1/30 border-white/[0.08]',
						].join(' ')}
						{...rest}
					>
						<View className="flex-1 flex-row items-baseline gap-1">
							<Text className="text-fg-0 font-sans-bold text-base">{set.reps}</Text>
							<Text className="text-fg-3 font-sans text-base">×</Text>
							<Text className="text-fg-0 font-sans-bold text-base">{set.kg}</Text>
							<Text className="text-fg-2 font-sans-semi text-[11px]">kg</Text>
						</View>
					</Pressable>
				)
			}}
		</form.Subscribe>
	)
}
