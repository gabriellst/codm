// packages/app/expo/app/active-exercise/-components/SetList/index.tsx
//
// Composes SetRow rows from server data, and renders a "draft preview"
// ghost row UNDER the list that mirrors the editor's live values.
//
// The draft preview is the lesson here: it's extracted into its own
// component `DraftSetPreviewSubscriber` precisely so that the keystroke-
// driven re-renders (one per character typed in NumField) stay scoped to
// that subscriber. If we inlined the <form.Subscribe> here, the FlatList
// would re-render on every keystroke — that's expensive and breaks scroll
// position on some platforms.

import { FlatList, Text, View, type ViewProps } from 'react-native'
import { useTranslation } from 'react-i18next'
import type { GetActiveExercise200 } from '@codedm/client-typescript/typescript'
import type { SetEditorForm } from '@/app/active-exercise/-types'
import { SetRow } from './SetRow'

type ActiveSet = GetActiveExercise200['sets'][number]

interface SetListProps extends Omit<ViewProps, 'children'> {
	form: SetEditorForm
	sets: ActiveSet[]
}

export function SetList({ form, sets, ...rest }: SetListProps) {
	return (
		<View className="flex-1" {...rest}>
			<FlatList
				data={sets}
				keyExtractor={item => item.workoutSetId}
				renderItem={({ item }) => (
					<View className="px-5 pb-1.5">
						<SetRow form={form} set={item} />
					</View>
				)}
				// Footer is its OWN component so the subscriber's re-renders
				// don't bubble back into FlatList.
				ListFooterComponent={<DraftSetPreviewSubscriber form={form} />}
			/>
		</View>
	)
}

/**
 * Renders a ghost row reflecting the editor's live values. Subscribes to
 * the four fields it cares about — not the whole form state. Returns null
 * in edit mode (no point ghosting a row that's already highlighted in the
 * list) and when reps/kg aren't numbers yet (defaults haven't loaded).
 */
function DraftSetPreviewSubscriber({ form }: { form: SetEditorForm }) {
	const { t } = useTranslation()
	return (
		<form.Subscribe
			selector={s => ({
				reps: s.values.reps,
				kg: s.values.kg,
				isDropset: s.values.isDropset,
				editing: s.values.editingSelection,
			})}
		>
			{({ reps, kg, isDropset, editing }) => {
				if (editing != null) return null
				if (typeof reps !== 'number' || typeof kg !== 'number') return null
				return (
					<View
						className="mx-5 mt-1 flex-row items-center gap-2.5 py-4 px-3.5"
						style={{
							opacity: 0.4,
							borderWidth: 1,
							borderStyle: 'dashed',
							borderColor: 'rgba(255,255,255,0.4)',
							borderRadius: 8,
						}}
					>
						<View className="flex-1 flex-row items-baseline gap-1">
							<Text className="text-fg-0 font-sans-bold text-base">{reps}</Text>
							<Text className="text-fg-3 font-sans text-base">×</Text>
							<Text className="text-fg-0 font-sans-bold text-base">{kg}</Text>
							<Text className="text-fg-2 font-sans-semi text-[11px]">kg</Text>
							{isDropset ? (
								<Text className="ml-1.5 text-accent-warning font-sans-bold text-[9px] uppercase">
									{t('workout.dropset')}
								</Text>
							) : null}
						</View>
						<Text className="text-fg-2 font-sans-bold text-[10px] uppercase">
							{t('workout.draftSet')}
						</Text>
					</View>
				)
			}}
		</form.Subscribe>
	)
}
