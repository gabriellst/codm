# P1 — Contracts Package Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `packages/contracts/` as the single source of truth for cross-language vocabulary: TypeSpec defines wire enums + integration events, Drizzle defines DB schema (text + jsonb, no pgEnum), and codegen emits TS/Rust/Go bindings into `packages/contracts/generated/{ts,rs,go}`.

**Architecture:** TypeSpec compiles to a single OpenAPI 3 YAML. Custom Bun emitters parse that YAML and produce per-language bindings (TS enums + Zod, Rust enums + serde, Go typed strings + structs). Drizzle is independent — text columns only, no pgEnum codegen needed. Build pipeline: `bun contracts` runs `tsp compile` + three emitters + `drizzle-kit generate` in sequence.

**Tech Stack:** `@typespec/compiler@^1.5`, `@typespec/openapi3@^1.5`, `drizzle-orm@^0.45`, `drizzle-kit@^0.31`, `yaml@^2.6`, Bun for codegen scripts.

Create `packages/contracts/codegen/lib/parse-openapi.ts`:

```typescript
import { parse as parseYaml } from 'yaml'

export type FieldType =
	| { kind: 'string' }
	| { kind: 'literal'; value: string }
	| { kind: 'enum-ref'; ref: string }
	| { kind: 'boolean' }
	| { kind: 'integer'; format?: 'int32' | 'int64' }
	| { kind: 'number'; format?: 'float' | 'float32' | 'float64' }
	| { kind: 'date-time' }
	| { kind: 'url' }
	| { kind: 'unknown' }

export type EventField = {
	name: string
	type: FieldType
	required: boolean
	doc?: string
}

export type ParsedEnum = { name: string; values: string[]; doc?: string }

export type ParsedEvent = {
	modelName: string
	wireName: string
	doc?: string
	fields: EventField[]
}

export type ParsedContracts = {
	enums: ParsedEnum[]
	events: ParsedEvent[]
}

interface OASchema {
	type?: string
	format?: string
	enum?: string[]
	properties?: Record<string, OASchema>
	required?: string[]
	allOf?: Array<{ $ref?: string } | OASchema>
	discriminator?: { propertyName: string }
	description?: string
	$ref?: string
}

interface OADoc {
	components?: { schemas?: Record<string, OASchema> }
}

const ENVELOPE = 'IntegrationEvent'

export function parseContractsOpenapi(yamlText: string): ParsedContracts {
	const doc = parseYaml(yamlText) as OADoc
	const schemas = doc.components?.schemas ?? {}

	const enums: ParsedEnum[] = []
	const events: ParsedEvent[] = []

	for (const [name, schema] of Object.entries(schemas)) {
		if (schema.type === 'string' && Array.isArray(schema.enum)) {
			enums.push({ name, values: schema.enum, doc: schema.description })
			continue
		}
		const ev = tryParseEvent(name, schema, schemas)
		if (ev) events.push(ev)
	}

	return { enums, events }
}

function tryParseEvent(modelName: string, schema: OASchema, all: Record<string, OASchema>): ParsedEvent | null {
	if (!Array.isArray(schema.allOf)) return null
	const extendsEnvelope = schema.allOf.some((s) => '$ref' in s && s.$ref?.endsWith(`/${ENVELOPE}`))
	if (!extendsEnvelope) return null

	const ownSchema = schema.allOf.find((s) => !('$ref' in s)) as OASchema | undefined
	if (!ownSchema) return null

	const envSchema = all[ENVELOPE]
	if (!envSchema) return null

	const requiredSet = new Set([...(envSchema.required ?? []), ...(ownSchema.required ?? [])])
	const props: Record<string, OASchema> = { ...(envSchema.properties ?? {}), ...(ownSchema.properties ?? {}) }

	// wireName comes from the `name` property's const-enum
	const nameProp = props.name
	const wireName = nameProp?.enum?.[0]
	if (!wireName) return null

	const fields: EventField[] = Object.entries(props).map(([fieldName, fs]) => ({
		name: fieldName,
		type: typeOf(fs),
		required: requiredSet.has(fieldName),
		doc: fs.description,
	}))

	return { modelName, wireName, doc: schema.description, fields }
}

function typeOf(s: OASchema): FieldType {
	if (s.$ref) {
		const name = s.$ref.split('/').pop()!
		return { kind: 'enum-ref', ref: name }
	}
	if (s.enum && s.enum.length === 1) return { kind: 'literal', value: s.enum[0]! }
	if (s.type === 'string') {
		if (s.format === 'date-time') return { kind: 'date-time' }
		if (s.format === 'uri') return { kind: 'url' }
		return { kind: 'string' }
	}
	if (s.type === 'integer') {
		const f = s.format
		return { kind: 'integer', format: f === 'int32' ? 'int32' : f === 'int64' ? 'int64' : undefined }
	}
	if (s.type === 'number') {
		const f = s.format
		return { kind: 'number', format: f === 'float' || f === 'float32' || f === 'float64' ? (f as 'float') : undefined }
	}
```

```typescript
const ENVELOPE_FIELDS = new Set(['name', 'entityId', 'ownerId', 'occurredAt'])

function zodExpr(t: FieldType): string {
	switch (t.kind) {
		case 'string':
			return 'z.string()'
		case 'literal':
			return `z.literal('${t.value}')`
		case 'enum-ref':
			return `${t.ref}Schema`
		case 'boolean':
			return 'z.boolean()'
		case 'integer':
			return 'z.number().int()'
		case 'number':
			return 'z.number()'
		case 'date-time':
			return 'z.string().datetime()'
		case 'url':
			return 'z.string().url()'
		case 'unknown':
			return 'z.unknown()'
	}
}

export function emitTsEvents(events: ParsedEvent[]): Record<string, string> {
	const files: Record<string, string> = {}

	for (const ev of events) {
		// Filter to payload fields only (drop envelope fields — the helper adds them)
		const payloadFields = ev.fields.filter((f) => !ENVELOPE_FIELDS.has(f.name))

		const enumRefs = payloadFields
			.filter((f) => f.type.kind === 'enum-ref')
			.map((f) => (f.type as { ref: string }).ref)
		const enumImports = [...new Set(enumRefs)].sort()
		const importLines = [`import { z } from '../_z'`]
		if (enumImports.length > 0) {
			const names = enumImports.flatMap((n) => [n, `${n}Schema`]).join(', ')
			importLines.push(`import { ${names} } from '../enums'`)
		}

		const zodFields = payloadFields
			.map((f) => `\t${f.name}: ${zodExpr(f.type)}${f.required ? '' : '.optional()'},`)
			.join('\n')

		const body =
			HEADER +
			importLines.join('\n') +
			'\n\n' +
			(ev.doc ? `/** ${ev.doc} */\n` : '') +
			`export const ${ev.modelName}Schema = z.integrationEvent('${ev.wireName}', {\n${zodFields}\n})\n\n` +
			`export type ${ev.modelName} = z.infer<typeof ${ev.modelName}Schema>\n`

		files[`${kebab(ev.modelName.replace(/Event$/, ''))}.ts`] = body
	}

	return files
}

export function emitTsBarrel(events: ParsedEvent[]): string {
	const names = events.map((e) => e.modelName)
	const reExports = events
		.map((e) => `export * from './${kebab(e.modelName.replace(/Event$/, ''))}'`)
		.sort()
		.join('\n')
	const namedImports = names.flatMap((n) => [`\t${n},`, `\t${n}Schema,`]).join('\n')
	const union = names.join(' | ')
	const schemas = names.map((n) => `\t${n}Schema`).join(',\n')
	return (
		HEADER +
		`import { z } from '../_z'\n` +
		`import {\n${namedImports}\n} from './_imports'\n\n` +
		reExports +
		'\n\n' +
		`export type IntegrationEvent = ${union}\n\n` +
		`export const IntegrationEventSchema = z.discriminatedUnion('name', [\n${schemas},\n])\n`
	)
}
```

```rust
//! Declarative macro for declaring an integration event struct.
//!
//! Expands `integration_event!("wire.name" => StructName { field: Type, ... })`
//! into a struct with envelope fields (entity_id, owner_id, occurred_at),
//! a `WIRE_NAME` constant, and serde derives with camelCase rename.
//!
//! In P3, this will be replaced by a richer proc-macro that takes the
//! `z.string()` / `z.int()` DSL (matching `feat/clean-rust`) and emits
//! the same struct plus Schema/ToSchema impls. For now the simpler
//! declarative form is sufficient — codegen produces raw Rust types.

#[macro_export]
macro_rules! integration_event {
    ($wire_name:literal => $name:ident { $($field_name:ident: $field_type:ty),* $(,)? }) => {
        #[derive(Debug, Clone, ::serde::Serialize, ::serde::Deserialize)]
        #[serde(rename_all = "camelCase")]
        pub struct $name {
            pub entity_id: ::std::string::String,
            pub owner_id: ::std::string::String,
            pub occurred_at: ::chrono::DateTime<::chrono::Utc>,
            $(pub $field_name: $field_type,)*
        }

        impl $name {
            pub const WIRE_NAME: &'static str = $wire_name;
        }
    };
}
```

```typescript
import { dirname, join } from 'node:path'
import { fileURLToPath } from 'node:url'
import { parseContractsOpenapi, type ParsedEnum, type ParsedEvent, type FieldType } from './lib/parse-openapi'

const HERE = dirname(fileURLToPath(import.meta.url))
const ROOT = join(HERE, '..')
const INPUT = join(ROOT, 'dist/contracts.openapi.yaml')
const OUTPUT = join(ROOT, 'generated/rs/src/wire')

const HEADER = `// AUTO-GENERATED by packages/contracts/codegen/emit-wire-rs.ts — do not edit.\n`

const ENVELOPE_FIELDS = new Set(['name', 'entityId', 'ownerId', 'occurredAt'])

function snake(name: string): string {
	return name.replace(/([a-z])([A-Z])/g, '$1_$2').toLowerCase()
}

function rsType(t: FieldType): string {
	switch (t.kind) {
		case 'string':
			return 'String'
		case 'literal':
			return 'String'
		case 'enum-ref':
			return `crate::wire::enums::${t.ref}`
		case 'boolean':
			return 'bool'
		case 'integer':
			return t.format === 'int64' ? 'i64' : 'i32'
		case 'number':
			return t.format?.includes('64') ? 'f64' : 'f32'
		case 'date-time':
			return 'chrono::DateTime<chrono::Utc>'
		case 'url':
			return 'String'
		case 'unknown':
			return 'serde_json::Value'
	}
}

export function emitRsEnums(enums: ParsedEnum[]): string {
	const lines: string[] = [HEADER]
	for (const e of enums) {
		lines.push('')
		if (e.doc) lines.push(`/// ${e.doc}`)
		lines.push('#[derive(')
		lines.push('\tDebug, Clone, Copy, PartialEq, Eq, Hash,')
		lines.push('\tserde::Serialize, serde::Deserialize,')
		lines.push('\tstrum::EnumString, strum::IntoStaticStr, strum::Display,')
		lines.push(')]')
		lines.push('#[allow(non_camel_case_types)]')
		lines.push(`pub enum ${e.name} {`)
		for (const v of e.values) lines.push(`\t${v},`)
		lines.push('}')
	}
	return lines.join('\n') + '\n'
}

export function emitRsEvents(events: ParsedEvent[]): string {
	const lines: string[] = [HEADER, 'use crate::integration_event;', '']
	for (const ev of events) {
		// Filter to payload fields (macro adds envelope)
		const payloadFields = ev.fields.filter((f) => !ENVELOPE_FIELDS.has(f.name))
		if (ev.doc) lines.push(`/// ${ev.doc}`)
		lines.push(`integration_event!("${ev.wireName}" => ${ev.modelName} {`)
		for (const f of payloadFields) {
			const ty = rsType(f.type)
			const wrapped = f.required ? ty : `Option<${ty}>`
			lines.push(`\t${snake(f.name)}: ${wrapped},`)
		}
		lines.push('});')
		lines.push('')
	}
	return lines.join('\n')
}

export function emitRsEnvelope(events: ParsedEvent[]): string {
	const lines: string[] = [HEADER, 'use super::events::*;', '']
	lines.push('#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]')
	lines.push('#[serde(tag = "name")]')
	lines.push('pub enum IntegrationEvent {')
	for (const ev of events) {
		const variant = ev.modelName.replace(/Event$/, '')
		lines.push(`\t#[serde(rename = "${ev.wireName}")]`)
		lines.push(`\t${variant}(${ev.modelName}),`)
	}
	lines.push('}')
	return lines.join('\n') + '\n'
}

async function run() {
	const yamlText = await readFile(INPUT, 'utf-8')
	const parsed = parseContractsOpenapi(yamlText)
	await rm(OUTPUT, { recursive: true, force: true })
	await mkdir(OUTPUT, { recursive: true })
	await writeFile(join(OUTPUT, 'enums.rs'), emitRsEnums(parsed.enums))
	await writeFile(join(OUTPUT, 'events.rs'), emitRsEvents(parsed.events))
	await writeFile(join(OUTPUT, 'envelope.rs'), emitRsEnvelope(parsed.events))
	await writeFile(
		join(OUTPUT, 'mod.rs'),
		HEADER + 'pub mod enums;\npub mod events;\npub mod envelope;\n',
	)
	// lib.rs at the crate root: declare the macro module BEFORE wire so the macro is in scope
	const libRs = HEADER + '#[macro_use]\npub mod integration_event;\npub mod wire;\n'
	await writeFile(join(OUTPUT, '..', 'lib.rs'), libRs)
	console.log(`✔ emitted Rust bindings: ${parsed.enums.length} enums, ${parsed.events.length} events`)
}

if (import.meta.main) {
	await run()
}
```

```typescript
import { mkdir, writeFile, rm, readFile } from 'node:fs/promises'
import { dirname, join } from 'node:path'
import { fileURLToPath } from 'node:url'
import { parseContractsOpenapi, type ParsedEnum, type ParsedEvent, type FieldType } from './lib/parse-openapi'

const HERE = dirname(fileURLToPath(import.meta.url))
const ROOT = join(HERE, '..')
const INPUT = join(ROOT, 'dist/contracts.openapi.yaml')
const OUTPUT = join(ROOT, 'generated/go/wire')

const HEADER = `// AUTO-GENERATED by packages/contracts/codegen/emit-wire-go.ts — do not edit.\n`

function pascalToGoField(name: string): string {
	// jsonName -> GoFieldName (e.g., videoId -> VideoID)
	const out = name[0]!.toUpperCase() + name.slice(1)
	return out.replace(/Id\b/g, 'ID').replace(/Url\b/g, 'URL')
}

function goType(t: FieldType): string {
	switch (t.kind) {
		case 'string':
			return 'string'
		case 'literal':
			return 'string'
		case 'enum-ref':
			return t.ref
		case 'boolean':
			return 'bool'
		case 'integer':
			return t.format === 'int64' ? 'int64' : 'int32'
		case 'number':
			return t.format?.includes('64') ? 'float64' : 'float32'
		case 'date-time':
			return 'time.Time'
		case 'url':
			return 'string'
		case 'unknown':
			return 'json.RawMessage'
	}
}

export function emitGoEnums(enums: ParsedEnum[]): string {
	const lines: string[] = [HEADER, 'package wire', '', 'import "fmt"', '']
	for (const e of enums) {
		lines.push(`type ${e.name} string`)
		lines.push('')
		lines.push('const (')
		for (const v of e.values) lines.push(`\t${e.name}${v} ${e.name} = "${v}"`)
		lines.push(')')
		lines.push('')
		lines.push(`func Parse${e.name}(s string) (${e.name}, error) {`)
		lines.push(`\tswitch ${e.name}(s) {`)
		const cases = e.values.map((v) => `${e.name}${v}`).join(', ')
		lines.push(`\tcase ${cases}:`)
		lines.push(`\t\treturn ${e.name}(s), nil`)
		lines.push('\tdefault:')
		lines.push(`\t\treturn "", fmt.Errorf("invalid ${e.name}: %q", s)`)
		lines.push('\t}')
		lines.push('}')
		lines.push('')
	}
	return lines.join('\n')
}

export function emitGoEvents(events: ParsedEvent[]): string {
	const needsTime = events.some((e) => e.fields.some((f) => f.type.kind === 'date-time'))
	const needsJson = events.some((e) => e.fields.some((f) => f.type.kind === 'unknown'))
	const lines: string[] = [HEADER, 'package wire', '']
	const imports: string[] = []
	if (needsTime) imports.push('"time"')
	if (needsJson) imports.push('"encoding/json"')
	if (imports.length > 0) {
		lines.push('import (')
		for (const i of imports) lines.push(`\t${i}`)
		lines.push(')')
		lines.push('')
	}

	for (const ev of events) {
		lines.push(`type ${ev.modelName} struct {`)
		for (const f of ev.fields) {
			if (f.name === 'name') continue
			const goField = pascalToGoField(f.name)
			const ty = goType(f.type)
			const ptr = f.required ? '' : '*'
			const tag = f.required ? f.name : `${f.name},omitempty`
			lines.push(`\t${goField} ${ptr}${ty} \`json:"${tag}"\``)
		}
		lines.push('}')
		lines.push('')
		// EventName()
		lines.push(`func (e ${ev.modelName}) EventName() string { return "${ev.wireName}" }`)
		lines.push('')
	}
	return lines.join('\n')
}

export function emitGoEnvelope(events: ParsedEvent[]): string {
	const lines: string[] = [HEADER, 'package wire', '', 'import (', '\t"encoding/json"', '\t"fmt"', ')', '']
	lines.push('type IntegrationEvent interface {')
	lines.push('\tEventName() string')
	lines.push('}')
	lines.push('')
	lines.push('func UnmarshalIntegrationEvent(data []byte) (IntegrationEvent, error) {')
	lines.push('\tvar discriminator struct { Name string `json:"name"` }')
	lines.push('\tif err := json.Unmarshal(data, &discriminator); err != nil { return nil, err }')
	lines.push('\tswitch discriminator.Name {')
	for (const ev of events) {
		lines.push(`\tcase "${ev.wireName}":`)
		lines.push(`\t\tvar v ${ev.modelName}`)
		lines.push('\t\tif err := json.Unmarshal(data, &v); err != nil { return nil, err }')
		lines.push('\t\treturn v, nil')
	}
	lines.push('\tdefault:')
	lines.push('\t\treturn nil, fmt.Errorf("unknown integration event: %q", discriminator.Name)')
	lines.push('\t}')
	lines.push('}')
	return lines.join('\n') + '\n'
}

async function run() {
	const yamlText = await readFile(INPUT, 'utf-8')
	const parsed = parseContractsOpenapi(yamlText)
	await rm(OUTPUT, { recursive: true, force: true })
	await mkdir(OUTPUT, { recursive: true })
	await writeFile(join(OUTPUT, 'enums.go'), emitGoEnums(parsed.enums))
	await writeFile(join(OUTPUT, 'events.go'), emitGoEvents(parsed.events))
	await writeFile(join(OUTPUT, 'envelope.go'), emitGoEnvelope(parsed.events))
	console.log(`✔ emitted Go bindings: ${parsed.enums.length} enums, ${parsed.events.length} events`)
}

if (import.meta.main) {
	await run()
}
```
